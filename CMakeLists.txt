cmake_minimum_required(VERSION 3.24)
project(AlbionAssistant VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use MSVC x64 para o aplicativo Windows")
endif()
include(scripts/msvc-dependencies.cmake)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/permissive-;/utf-8;/EHsc>")
configure_file(resources/app.rc.in app.rc @ONLY)
set(APP_RESOURCES "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
set_property(SOURCE "${APP_RESOURCES}" PROPERTY OBJECT_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/resources/app.ico"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/assassin-none.png"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/assassin-2.png"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/assassin-3.png"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/assassin-clock.png")
include(CTest)
add_library(assistant_rules src/rules.cpp src/settings.cpp src/profiles.cpp src/workspace.cpp src/monitor.cpp)
target_include_directories(assistant_rules PUBLIC src)
add_library(assistant_vision src/recognition.cpp src/image.cpp src/calibration.cpp src/overlay_effect.cpp)
target_include_directories(assistant_vision PUBLIC src)
target_link_libraries(assistant_vision PUBLIC windowscodecs ole32)
add_library(assistant_windows src/capture.cpp src/overlay.cpp src/selection.cpp)
target_link_libraries(assistant_windows PUBLIC assistant_rules assistant_vision d3d11 dxgi user32 gdi32 ole32 comctl32 comdlg32 shell32)
add_library(assistant_app src/app.cpp src/app_ui.cpp src/theme.cpp)
target_link_libraries(assistant_app PUBLIC assistant_windows uxtheme dwmapi)
if(BUILD_TESTING)
    add_test(NAME build_dependencies COMMAND "${CMAKE_COMMAND}" -E env --unset=PSModulePath powershell.exe -NoProfile -ExecutionPolicy Bypass -File
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/build_dependencies_tests.ps1" -CMake "${CMAKE_COMMAND}"
        -Ninja "${CMAKE_MAKE_PROGRAM}" -Compiler "${CMAKE_CXX_COMPILER}" -BuildDirectory "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(build_dependencies PROPERTIES TIMEOUT 120)
    add_executable(rules_tests tests/rules_tests.cpp)
    target_link_libraries(rules_tests PRIVATE assistant_rules)
    add_test(NAME rules COMMAND rules_tests)
    add_executable(recognition_tests tests/recognition_tests.cpp "${APP_RESOURCES}")
    target_link_libraries(recognition_tests PRIVATE assistant_vision)
    add_test(NAME recognition COMMAND recognition_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
    set_tests_properties(recognition PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(profiles_tests tests/profiles_tests.cpp)
    target_link_libraries(profiles_tests PRIVATE assistant_rules)
    add_test(NAME profiles COMMAND profiles_tests)
    add_executable(workspace_tests tests/workspace_tests.cpp)
    target_link_libraries(workspace_tests PRIVATE assistant_rules)
    add_test(NAME workspace COMMAND workspace_tests)
    add_executable(monitor_tests tests/monitor_tests.cpp)
    target_link_libraries(monitor_tests PRIVATE assistant_rules)
    add_test(NAME monitor COMMAND monitor_tests)
    add_executable(calibration_tests tests/calibration_tests.cpp)
    target_link_libraries(calibration_tests PRIVATE assistant_vision)
    add_test(NAME calibration COMMAND calibration_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
    set_tests_properties(calibration PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(selection_tests tests/selection_tests.cpp)
    target_link_libraries(selection_tests PRIVATE assistant_vision user32 gdi32)
    add_test(NAME selection COMMAND selection_tests)
    set_tests_properties(selection PROPERTIES TIMEOUT 20)
    add_executable(overlay_effect_tests tests/overlay_effect_tests.cpp)
    target_link_libraries(overlay_effect_tests PRIVATE assistant_vision user32 gdi32)
    add_test(NAME overlay_effect COMMAND overlay_effect_tests)
    add_executable(panel_layout_tests tests/panel_layout_tests.cpp)
    target_include_directories(panel_layout_tests PRIVATE src)
    add_test(NAME panel_layout COMMAND panel_layout_tests)
    add_executable(app_flow_tests tests/app_flow_tests.cpp "${APP_RESOURCES}")
    target_link_libraries(app_flow_tests PRIVATE assistant_app)
    add_test(NAME app_flow COMMAND app_flow_tests)
    set_tests_properties(app_flow PROPERTIES TIMEOUT 30)
    add_executable(packaging_tests tests/packaging_tests.cpp)
    target_link_libraries(packaging_tests PRIVATE version)
    add_test(NAME packaging COMMAND packaging_tests "$<TARGET_FILE:AlbionAssistant>")
    # CTest pode herdar módulos do PowerShell 7; o Windows PowerShell precisa dos próprios módulos.
    add_test(NAME build_script COMMAND "${CMAKE_COMMAND}" -E env --unset=PSModulePath powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/build_script_tests.ps1")
    add_test(NAME delivery COMMAND "${CMAKE_COMMAND}" -E env --unset=PSModulePath powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/delivery_tests.ps1")
endif()
add_executable(AlbionAssistant WIN32 src/main.cpp resources/app.manifest "${APP_RESOURCES}")
target_link_libraries(AlbionAssistant PRIVATE assistant_app)
install(TARGETS AlbionAssistant RUNTIME DESTINATION .)
